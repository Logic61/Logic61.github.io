Python 实现图片压缩

这些结果可以用于图片压缩。以下是利用奇异值分解实现灰度图片压缩的代码示例：

import numpy as np
import matplotlib.pyplot as plt
from PIL import Image

# 1. 加载图片并转换为灰度矩阵
img = Image.open('your_image_path.png').convert('L')
A = np.array(img)

# 2. 进行奇异值分解
U, s, Vt = np.linalg.svd(A)

# 3. 取前 k 个奇异值进行重构
def reconstruct(k):
    Sigma_k = np.diag(s[:k])
    A_k = U[:, :k] @ Sigma_k @ Vt[:k, :]
    return A_k

# 4. 可视化对比
plt.figure(figsize=(12, 4))
for i, k in enumerate([5, 20, 100]):
    plt.subplot(1, 3, i+1)
    plt.imshow(reconstruct(k), cmap='gray')
    plt.title(f'Top {k} Singular Values')
plt.show()
